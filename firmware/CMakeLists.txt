cmake_minimum_required(VERSION 3.16)

# The commit this image is built from (spec 106 AC-2): PANELATOR_COMMIT when set, else git, else an error.
if(DEFINED ENV{PANELATOR_COMMIT} AND NOT "$ENV{PANELATOR_COMMIT}" STREQUAL "")
  if(NOT "$ENV{PANELATOR_COMMIT}" MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "PANELATOR_COMMIT must be 40 lower-case hex digits, got '$ENV{PANELATOR_COMMIT}'")
  endif()
  string(LENGTH "$ENV{PANELATOR_COMMIT}" _commit_len)
  if(NOT _commit_len EQUAL 40)
    message(FATAL_ERROR "PANELATOR_COMMIT must be 40 hex digits, got ${_commit_len}")
  endif()
  set(PANELATOR_COMMIT "$ENV{PANELATOR_COMMIT}")
else()
  execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}"
                  OUTPUT_VARIABLE _head OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE _head_rc ERROR_QUIET)
  if(NOT _head_rc EQUAL 0 OR _head STREQUAL "")
    message(FATAL_ERROR "Cannot tell which commit this is: set PANELATOR_COMMIT to the 40 hex digits of the commit, or build inside a git checkout")
  endif()
  execute_process(COMMAND git diff-index --quiet HEAD -- WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}" RESULT_VARIABLE _dirty_rc ERROR_QUIET)
  if(_dirty_rc EQUAL 0)
    set(PANELATOR_COMMIT "${_head}")
  else()
    set(PANELATOR_COMMIT "${_head}-dirty")
  endif()
endif()
message(STATUS "panelator commit: ${PANELATOR_COMMIT}")

include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(panelator)
